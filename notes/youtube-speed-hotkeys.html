FIELD NOTES NOTE 004 // YOUTUBE / TAMPERMONKEY

YouTube playback-speed hotkeys

A tiny userscript for immediate playback control: 0.25×, 1×, 2×, 3× and 4× from the keyboard, plus temporary 2× while holding Q.

In this note
01Install Tampermonkey02Create a new userscript03Save and reload YouTubeSummary
What this solves

No menus and no repeated clicks. The script changes the playback rate of the current YouTube video and ignores the hotkeys while you are typing in search, comments or other text fields.

01

Install Tampermonkey

Install the Tampermonkey browser extension. In Opera, Chrome or Edge, open the extension settings and make sure Allow UserScripts / UserScripts zulassen is enabled. Without that switch, the userscript will not run.

02

Create a new userscript

Open Tampermonkey and choose Create a new script / Neues Skript erstellen. Delete the default content and paste the current script below.

// ==UserScript==
// @name         YouTube Speed Hotkeys Fixed
// @match        https://www.youtube.com/*
// @match        https://youtube.com/*
// @grant        none
// @run-at       document-idle
// ==/UserScript==

(function () {
  function getVideo() {
    return document.querySelector("video");
  }

  function isTyping() {
    const el = document.activeElement;
    if (!el) return false;

    const tag = el.tagName?.toLowerCase();
    return (
      tag === "input" ||
      tag === "textarea" ||
      el.isContentEditable
    );
  }

  function setSpeed(rate) {
    const video = getVideo();
    if (!video) return;

    video.playbackRate = rate;
    console.log("YouTube speed set to:", rate);
  }

  window.addEventListener("keydown", function (e) {
    if (isTyping()) return;

    if (e.key === "2") {
      e.preventDefault();
      e.stopPropagation();
      setSpeed(2);
    }

    if (e.key === "0") {
      e.preventDefault();
      e.stopPropagation();
      setSpeed(0.25);
    }

    if (e.key === "1") {
      e.preventDefault();
      e.stopPropagation();
      setSpeed(1);
    }

    if (e.key === "3") {
      e.preventDefault();
      e.stopPropagation();
      setSpeed(3);
    }

    if (e.key === "4") {
      e.preventDefault();
      e.stopPropagation();
      setSpeed(4);
    }

    if (e.key.toLowerCase() === "q") {
      e.preventDefault();
      e.stopPropagation();
      setSpeed(2);
    }
  }, true);

  window.addEventListener("keyup", function (e) {
    if (isTyping()) return;

    if (e.key.toLowerCase() === "q") {
      e.preventDefault();
      e.stopPropagation();
      setSpeed(1);
    }
  }, true);
})();
03

Save and reload YouTube

Save the userscript with Ctrl + S. Open or reload YouTube, then start any video.

Hotkeys

  • 0 → 0.25× playback.
  • 1 → normal 1× playback.
  • 2 → fixed 2× playback.
  • 3 → fixed 3× playback.
  • 4 → fixed 4× playback.
  • Hold Q → temporary 2×.
  • Release Q → return to 1×.
  • Typing fields are ignored, so the shortcuts do not interfere with search or comments.
