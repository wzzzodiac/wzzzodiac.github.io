FIELD NOTES NOTE 004 // YOUTUBE / TAMPERMONKEY

YouTube playback hotkeys

A small Tampermonkey userscript for fixed 1×–4× playback, pause and resume, time jumps and adaptive volume steps.

In this note
01Install Tampermonkey02Create a new userscript03Save and reload YouTubeSummary
What this solves

Control the current YouTube video from the keyboard, with smaller volume steps near silence. The hotkeys are ignored while you are typing in search, comments or other text fields.

01

Install Tampermonkey

Install the Tampermonkey browser extension. In Opera, Chrome or Edge, open the extension settings and make sure Allow UserScripts / UserScripts zulassen is enabled. Without that switch, the userscript will not run.

02

Create a new userscript

Open Tampermonkey and choose Create a new script / Neues Skript erstellen. Replace the entire editor content with version 2.2 below, including its metadata header. If you are updating an existing script, replace its full contents and disable any older duplicate copies.

// ==UserScript==
// @name         YouTube Speed Hotkeys Fixed
// @version      2.2
// @description  1-4 velocidad, 5 pausa, 6 seguir, 7 -10s, 8 +1min, 9 +10s, 0 -1min, -/+ volumen adaptativo
// @match        https://www.youtube.com/*
// @match        https://youtube.com/*
// @grant        unsafeWindow
// @sandbox      JavaScript
// @run-at       document-start
// ==/UserScript==

(function () {
    'use strict';

    function clamp(value, minimum, maximum) {
        return Math.max(minimum, Math.min(maximum, value));
    }

    // Pasos de 1 punto entre 0 y 10%; de 10 puntos por encima.
    function nextVolumePercent(volume, delta) {
        const percent = clamp(Math.round(volume * 100), 0, 100);
        if (delta > 0) {
            return percent < 10 ? percent + 1 : Math.min(100, percent + 10);
        }
        if (delta < 0) {
            return percent <= 10 ? Math.max(0, percent - 1) : Math.max(10, percent - 10);
        }
        return percent;
    }

    function commandFromKey(key) {
        const fixedRates = {
            '1': 1,
            '2': 2,
            '3': 3,
            '4': 4
        };

        if (fixedRates[key] !== undefined) {
            return {
                type: 'set-rate',
                rate: fixedRates[key]
            };
        }

        if (key === '+' || key === '=') {
            return {
                type: 'adjust-volume',
                delta: 0.1
            };
        }

        if (key === '-') {
            return {
                type: 'adjust-volume',
                delta: -0.1
            };
        }

        if (key === '5') {
            return {
                type: 'playback-state',
                state: 'pause'
            };
        }

        if (key === '6') {
            return {
                type: 'playback-state',
                state: 'play'
            };
        }

        if (key === '0') {
            return {
                type: 'seek',
                seconds: -60
            };
        }

        if (key === '7') {
            return {
                type: 'seek',
                seconds: -10
            };
        }

        if (key === '8') {
            return {
                type: 'seek',
                seconds: 60
            };
        }

        if (key === '9') {
            return {
                type: 'seek',
                seconds: 10
            };
        }

        return null;
    }

    function isTyping(event) {
        return event.composedPath().some(element =>
            ['INPUT', 'TEXTAREA', 'SELECT'].includes(element.tagName) ||
            element.isContentEditable
        ) || document.activeElement?.isContentEditable;
    }

    function getVideo() {
        // Prioriza el reproductor principal sobre las vistas previas.
        return document.querySelector('#movie_player video') ||
            document.querySelector('video');
    }

    const pageWindow = typeof unsafeWindow !== 'undefined' ? unsafeWindow : window;
    let statusTimer;
    let volumeCommandId = 0;

    function showStatus(text, error = false) {
        const host = document.fullscreenElement || document.body || document.documentElement;
        if (!host) return;
        let status = document.getElementById('yt-hotkeys-status');
        if (!status) {
            status = document.createElement('div');
            status.id = 'yt-hotkeys-status';
            status.setAttribute('role', 'status');
            status.dataset.ytHotkeysVersion = '2.2';
            status.style.cssText = 'position:fixed;top:80px;right:20px;z-index:2147483647;padding:10px 14px;border-radius:8px;background:#161616ee;color:#fff;font:14px/1.4 Arial,sans-serif;pointer-events:none;max-width:420px;box-shadow:0 2px 12px #0008';
        }
        if (status.parentNode !== host) host.appendChild(status);
        status.style.border = `1px solid ${error ? '#ff8888' : '#69d4bc'}`;
        status.textContent = `YT 2.2 · ${text}`;
        clearTimeout(statusTimer);
        statusTimer = setTimeout(() => status.remove(), 4500);
    }

    function getPagePlayer(video) {
        // Pedir explícitamente el objeto del mundo JavaScript de la página.
        // Un elemento DOM de un contexto aislado puede no exponer setVolume.
        const player = pageWindow.document.getElementById('movie_player');
        if (player?.contains(video)) return player;
        return null;
    }

    function adjustVolume(video, delta) {
        const commandId = ++volumeCommandId;
        const player = getPagePlayer(video);
        if (typeof player?.setVolume !== 'function' || typeof player?.getVolume !== 'function') {
            showStatus('sin conexión al control de YouTube. No se ha cambiado el volumen.', true);
            console.warn('YT Hotkeys 2.2: falta acceso a getVolume/setVolume en el contexto de página.');
            return;
        }

        const percent = player.getVolume();
        if (!Number.isFinite(percent)) {
            showStatus('YouTube todavía no informa de su volumen. Prueba otra vez.', true);
            return;
        }
        const muted = typeof player.isMuted === 'function' ? player.isMuted() : video.muted;
        const nextPercent = nextVolumePercent(muted ? 0 : clamp(percent, 0, 100) / 100, delta);

        // Se cambia el estado del reproductor, su barra y su valor recordado.
        // A cero se activa también silencio para que el vídeo quede silenciado.
        player.setVolume(nextPercent);
        if (nextPercent === 0) {
            if (typeof player.mute === 'function') player.mute();
        } else if (muted) {
            if (typeof player.unMute !== 'function') {
                showStatus('YouTube no permite quitar el silencio ahora.', true);
                return;
            }
            player.unMute();
            player.setVolume(nextPercent);
        }
        showStatus(`volumen solicitado: ${nextPercent}%`);

        // Solo se lee el resultado: no se bloquea el control manual del usuario.
        setTimeout(() => {
            if (commandId !== volumeCommandId || getVideo() !== video) return;
            try {
                const actual = player.getVolume();
                const slider = player.querySelector('.ytp-volume-panel[role="slider"]');
                const sliderValue = slider?.getAttribute('aria-valuenow');
                const sliderPercent = sliderValue === null || sliderValue === undefined
                    ? null : Number(sliderValue);
                const confirmed = Number.isFinite(actual) && Math.abs(actual - nextPercent) < 0.5 &&
                    (sliderPercent === null || Math.abs(sliderPercent - nextPercent) < 0.5);
                const bar = sliderPercent === null ? '' : ` · barra: ${sliderPercent}%`;
                showStatus(`pedido: ${nextPercent}% · YouTube: ${actual}%${bar}${confirmed ? ' ✓' : ' · sin confirmar'}`, !confirmed);
            } catch (error) {
                showStatus('no se pudo verificar el volumen de YouTube.', true);
                console.debug('YT Hotkeys 2.2: error al verificar volumen', error);
            }
        }, 1200);
    }

    window.addEventListener('keydown', event => {
        if (event.isComposing || isTyping(event) || event.ctrlKey || event.metaKey || event.altKey) {
            return;
        }

        const command = commandFromKey(event.key);
        const video = getVideo();
        if (!command || !video) return;

        // Impide que YouTube use los números para saltar a un porcentaje.
        event.preventDefault();
        event.stopImmediatePropagation();

        try {
            switch (command.type) {
                case 'set-rate':
                    video.playbackRate = command.rate;
                    break;
                case 'adjust-volume': {
                    adjustVolume(video, command.delta);
                    break;
                }
                case 'seek': {
                    const target = Math.max(0, video.currentTime + command.seconds);
                    video.currentTime = Number.isFinite(video.duration)
                        ? Math.min(target, video.duration)
                        : target;
                    break;
                }
                case 'playback-state':
                    if (command.state === 'pause') {
                        video.pause();
                    } else {
                        const promise = video.play();
                        if (promise?.catch) promise.catch(error => {
                            console.debug('YouTube: no se pudo reanudar el video', error);
                        });
                    }
                    break;
            }
        } catch (error) {
            if (command.type === 'adjust-volume') showStatus('YouTube rechazó el cambio de volumen.', true);
            console.debug('YouTube: no se pudo aplicar el control', error);
        }
    }, true);
})();
03

Save and reload YouTube

Save the userscript with Ctrl + S. Reload YouTube, then start a video. Use - or + to check the YT 2.2 status badge: it shows the requested volume and checks YouTube’s reported volume and its slider when available. If the player controls cannot be reached, the badge reports the problem.

Hotkeys

  • 1 → normal 1× playback.
  • 2 → fixed 2× playback.
  • 3 → fixed 3× playback.
  • 4 → fixed 4× playback.
  • 5 → pause.
  • 6 → resume playback.
  • 7 → back 10 seconds.
  • 8 → forward 1 minute.
  • 9 → forward 10 seconds.
  • 0 → back 1 minute.
  • - / + (or =) → lower / raise volume: 1 percentage point per step in the 0–10% range, 10 points above it. At zero, the player is muted; raising volume unmutes it.
  • Typing fields, IME input and shortcuts with Ctrl, Alt or Command are ignored.
