const VERSION = '0.1.0';
const PREFIX = 'wzc-instagram-readonly:v1:';
const TIMING = Object.freeze({ count: 50, maxPages: 250, longEvery: 15, retryLimit: 3 });

function cookieValue(cookie, name) {
  return cookie.split(';').map(part => part.trim()).find(part => part.startsWith(name + '='))?.slice(name.length + 1) || '';
}

function accountFromCookie(cookie) {
  const id = cookieValue(cookie, 'ds_user_id');
  if (!/^\d+$/.test(id)) throw new Error('Open Instagram and sign in before pasting the script.');
  return id;
}

function normalizeUser(raw) {
  const id = raw?.pk_id ?? raw?.pk ?? raw?.id;
  // Do not silently round numeric IDs, or treat malformed rows as missing followers.
  if ((typeof id === 'number' && !Number.isSafeInteger(id)) || !/^\d+$/.test(String(id))) {
    throw new Error('Instagram returned an account without a valid ID.');
  }
  if (typeof raw.username !== 'string' || !/^[a-zA-Z0-9._]{1,30}$/.test(raw.username)) {
    throw new Error('Instagram returned an unexpected username.');
  }
  return { id: String(id), username: raw.username, name: typeof raw.full_name === 'string' ? raw.full_name : (raw.name || '') };
}

function difference(users, other) {
  const ids = new Set(other.map(user => user.id));
  return users.filter(user => !ids.has(user.id));
}

function makeScan(followers, following, previous, completedAt = new Date().toISOString()) {
  return {
    completedAt,
    previousAt: previous?.completedAt || null,
    followers,
    followingCount: following.length,
    notFollowingBack: difference(following, followers),
    departed: previous ? difference(previous.followers, followers) : [],
  };
}

function openStore(storage, accountId) {
  const key = PREFIX + accountId;
  function read() {
    const raw = storage.getItem(key);
    if (!raw) return { version: 1, accountId, whitelist: [], scan: null, history: [] };
    let data;
    try {
      data = JSON.parse(raw);
      if (data.version !== 1 || data.accountId !== accountId || !Array.isArray(data.whitelist)) throw new Error();
      data.whitelist = data.whitelist.map(normalizeUser);
      data.history = (data.history || []).filter(entry => Number.isFinite(Date.parse(entry.at))).slice(0, 30);
      if (data.scan) {
        if (!Number.isFinite(Date.parse(data.scan.completedAt)) || !Number.isInteger(data.scan.followingCount)) throw new Error();
        for (const field of ['followers', 'notFollowingBack', 'departed']) data.scan[field] = data.scan[field].map(normalizeUser);
      }
    } catch {
      throw new Error('The cache for this account cannot be read. It has not been overwritten.');
    }
    return data;
  }
  function write(data) {
    try { storage.setItem(key, JSON.stringify(data)); }
    catch { throw new Error('Could not save in this browser. The previous baseline is preserved.'); }
    return data;
  }
  return {
    read,
    setWhitelist(user, included) {
      const data = read();
      data.whitelist = data.whitelist.filter(item => item.id !== user.id);
      if (included) data.whitelist.push(normalizeUser(user));
      return write(data);
    },
    saveScan(scan, expectedPreviousAt) {
      const data = read();
      if ((data.scan?.completedAt || null) !== expectedPreviousAt) throw new Error('Another scan updated the baseline. Please scan again.');
      data.scan = scan;
      data.history = [{ at: scan.completedAt, followers: scan.followers.length, following: scan.followingCount }, ...data.history].slice(0, 30);
      return write(data);
    },
  };
}

function abortableSleep(ms, signal) {
  return new Promise((resolve, reject) => {
    if (signal.aborted) return reject(new DOMException('Scan stopped.', 'AbortError'));
    const timer = setTimeout(() => { signal.removeEventListener('abort', abort); resolve(); }, ms);
    function abort() { clearTimeout(timer); signal.removeEventListener('abort', abort); reject(new DOMException('Scan stopped.', 'AbortError')); }
    signal.addEventListener('abort', abort, { once: true });
  });
}

async function scanAccount({ accountId, getCookie, fetcher, signal, onProgress = () => {}, sleep = abortableSleep, random = Math.random }) {
  let requests = 0;
  function guard() {
    if (signal.aborted) throw new DOMException('Scan stopped.', 'AbortError');
    if (accountFromCookie(getCookie()) !== accountId) throw new Error('The session changed. Close this panel and paste the script in the correct account.');
  }
  async function wait(ms, label) {
    guard();
    onProgress({ type: 'wait', ms, label });
    await sleep(ms, signal);
    guard();
  }
  async function request(kind, cursor) {
    // This is the only network call in the application: two fixed GET endpoints.
    const url = new URL(`https://www.instagram.com/api/v1/friendships/${accountId}/${kind}/`);
    url.searchParams.set('count', String(TIMING.count));
    if (cursor) url.searchParams.set('max_id', cursor);
    for (let retry = 0; ; retry++) {
      guard();
      if (requests > 0) {
        // Preserve the original scan pacing: 300–1000 ms + 1000–1300 ms.
        await wait(Math.floor(700 * random()) + 300 + Math.floor(300 * random()) + 1000, 'Waiting between requests');
        if (requests % TIMING.longEvery === 0) await wait(Math.round(10000 + (10000 * random() - 5000)), 'Long scan pause');
      }
      guard();
      const headers = { 'X-IG-App-ID': '936619743392459', 'X-ASBD-ID': '129477', 'X-Requested-With': 'XMLHttpRequest', Accept: '*/*' };
      const csrf = cookieValue(getCookie(), 'csrftoken');
      if (csrf) headers['X-CSRFToken'] = csrf;
      requests++;
      let response;
      try {
        response = await fetcher(url.href, { method: 'GET', credentials: 'same-origin', redirect: 'error', headers, signal });
      } catch (error) {
        guard();
        if (retry >= TIMING.retryLimit) throw new Error('Could not connect to Instagram. The previous baseline is preserved.');
        await wait(5000 * (retry + 1), 'Waiting for the connection');
        continue;
      }
      guard();
      if (response.status === 429) {
        if (retry >= TIMING.retryLimit) throw new Error('Instagram is still limiting requests. Please try later.');
        const retryHeader = response.headers?.get('retry-after');
        const serverWait = retryHeader ? (/^\d+$/.test(retryHeader) ? Number(retryHeader) * 1000 : Math.max(0, Date.parse(retryHeader) - Date.now())) : 0;
        const cooldown = Math.max(30000 * (retry + 1), serverWait || 0);
        if (cooldown > 300000) throw new Error('Instagram requires a long wait. Please try later.');
        await wait(cooldown, 'Instagram requests a pause');
        continue;
      }
      if (!response.ok) throw new Error(`Instagram returned HTTP ${response.status}. Scan stopped; open Instagram to check your session.`);
      let page;
      try { page = await response.json(); }
      catch { throw new Error('Instagram did not return a valid list. The previous baseline is preserved.'); }
      guard();
      if (!page || page.status === 'fail' || page.challenge || page.checkpoint_url || !Array.isArray(page.users)) {
        throw new Error('Instagram did not allow the request to complete. Check your session before trying again.');
      }
      return page;
    }
  }
  async function collect(kind) {
    const users = new Map();
    const seenCursors = new Set();
    let cursor = '';
    for (let index = 0; index < TIMING.maxPages; index++) {
      guard();
      onProgress({ type: 'page', kind, count: users.size, page: index + 1 });
      const page = await request(kind, cursor);
      const before = users.size;
      for (const raw of page.users) { const user = normalizeUser(raw); users.set(user.id, user); }
      const next = page.next_max_id == null ? '' : String(page.next_max_id);
      if (page.has_more === true && !next) throw new Error('The list is incomplete: Instagram did not provide the next page.');
      if (next && (seenCursors.has(next) || users.size === before)) throw new Error('Instagram repeated a page. The scan stopped to preserve the previous baseline.');
      if (!next) return [...users.values()];
      seenCursors.add(next);
      cursor = next;
    }
    throw new Error('The page limit was reached before completing the list. The previous baseline is preserved.');
  }
  const followers = await collect('followers');
  const following = await collect('following');
  guard();
  return { followers, following, requests };
}
